    ntestiterations = 20;
    DATA = ex_full.p1.DATA;
    [nx,ny,nz,nfile] = size(DATA);
    mask3d = ex_full.p1.mask3d;
    for tl=1:ntestiterations
      tic
      phansignal = zeros(2,nz,nfile); % 1 - mean; 2 - std
      % original compute method. for each slice of each timepoint
      for i=1:nfile
	  for ll=1:nz
	      % testing to confirm get same values
	      dataslice1 = DATA(:,:,ll,i);
	      maskphan1 = mask3d(:,:,ll); Iphan1 = maskphan1==1;
	      phansignal(1,ll,i) = mean(dataslice1(Iphan1)); phansignal(2,ll,i) = std(dataslice1(Iphan1));
	  end
      end
      double_loop_time(tl) = toc;

      % vectorized with deconstructed mean and std to use mask vox count
      tic
      [phan_m, phan_sd] = vec_mu_sigma(DATA,mask3d,[1 2]);
      vectorized_time(tl) = toc;
    end

    % ~3-4x faster than double loop
    vectorized_vs_loop =  median(double_loop_time)/median(vectorized_time)
  #+end_src

    fprintf('median time(n=%d): loop %.02fx slower then vectorized (%.2fs/%.2fs) for ~10^%.1f total elements (z=%d)\n',...
